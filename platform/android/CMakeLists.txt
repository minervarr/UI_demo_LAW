# ---------------------------------------------------------------------------
# ui_demo — Android shared library.
#
# A thin wrapper: it compiles the SAME src/app sources the desktop CMakeLists
# builds, and links app_shell's Android host instead of its Wayland one. That
# is the whole of the port — nothing under src/ knows which platform it is in.
# ---------------------------------------------------------------------------
cmake_minimum_required(VERSION 3.22.1)
project("ui_demo_android")

set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)

# AGP sets CMAKE_SOURCE_DIR to this directory. Probe upward so REPO_ROOT lands
# at the project root regardless of which CMakeLists.txt cmake calls top-level.
set(_probe ${CMAKE_SOURCE_DIR})
while(NOT EXISTS ${_probe}/libs/firstparty/vk_canvas/core/CMakeLists.txt)
    get_filename_component(_probe "${_probe}" DIRECTORY)
    if(_probe STREQUAL "")
        message(FATAL_ERROR
            "Cannot find the project root. Run: git submodule update --init --recursive")
    endif()
endwhile()
set(REPO_ROOT ${_probe})

set(VK_CANVAS     ${REPO_ROOT}/libs/firstparty/vk_canvas)
set(VK_CANVAS_DIR ${VK_CANVAS} CACHE PATH "vk_canvas root")
set(APP_SHELL_DIR ${REPO_ROOT}/libs/firstparty/app_shell)

# ── Shaders ─────────────────────────────────────────────────────────────────
# Compiled into the module's own assets/, which AGP packages. msdf_vert/frag
# are required because this app calls Renderer::initMsdf.
include(${VK_CANVAS}/cmake/VceShaders.cmake)
set(SHADER_OUT ${CMAKE_CURRENT_SOURCE_DIR}/app/src/main/assets/shaders)
vce_compile_slang(compile_shared_shaders ${SHADER_OUT}
    ${VK_CANVAS}/first_party/vulkan_font_engine/shaders_src
    composite_vert composite_frag tiling coverage msdf_vert msdf_frag)
vce_compile_slang(compile_canvas_shaders ${SHADER_OUT}
    ${VK_CANVAS}/shaders_src
    overlay_vert overlay_frag image_vert image_frag shape_vert shape_frag)

# ── ui_min_text_size.gen.h ──────────────────────────────────────────────────
# app_shell's ui_metrics.hh includes this unconditionally, and the generator
# that normally emits it is a HOST tool that measures the app's own faces — it
# cannot run cross-compiled. The desktop build computes the real number from
# the three NewComputerModern faces this app draws with; that value is stated
# here so the two agree. Change the desktop font list and change this too.
set(_gen_dir ${CMAKE_BINARY_DIR}/generated)
file(MAKE_DIRECTORY ${_gen_dir})
file(WRITE ${_gen_dir}/ui_min_text_size.gen.h
"#pragma once\n"
"// GENERATED by platform/android/CMakeLists.txt.\n"
"// Mirrors what app_shell_generate_min_text_size() computes on the desktop\n"
"// build from NewCM10 Regular/Bold/Italic - the host tool cannot run here.\n"
"constexpr float kMinReadableTextSizePx = 18.2857f;\n")

# ── app_shell (Android) ─────────────────────────────────────────────────────
set(APP_SHELL_APP_NAME     "ui_demo" CACHE STRING "" FORCE)
set(APP_SHELL_WINDOW_TITLE "vk_canvas demo" CACHE STRING "" FORCE)
set(APP_SHELL_APP_ID       "io.nava.uidemo" CACHE STRING "" FORCE)
set(APP_SHELL_EXTRA_INCLUDE ${_gen_dir})
set(APP_SHELL_NATIVE_APP_GLUE_DIR ${ANDROID_NDK}/sources/android/native_app_glue)
add_subdirectory(${APP_SHELL_DIR} ${CMAKE_BINARY_DIR}/app_shell)
target_include_directories(app_shell PUBLIC ${_gen_dir})

# ── vk_canvas + mathcore ────────────────────────────────────────────────────
add_subdirectory(${VK_CANVAS}/core   ${CMAKE_BINARY_DIR}/vk_canvas_core)
add_subdirectory(${REPO_ROOT}/libs/mathcore ${CMAKE_BINARY_DIR}/mathcore)

# ── The application shared library ──────────────────────────────────────────
# The source list is the desktop's UI_DEMO_APP_SRC plus this platform's entry
# point. Kept in step by hand; the desktop file names the same files.
add_library(ui_demo SHARED
    src/main.cc
    ${REPO_ROOT}/src/app/demo_app.cc
    ${REPO_ROOT}/src/app/page_text.cpp
    ${REPO_ROOT}/src/app/page_shapes.cpp
    ${REPO_ROOT}/src/app/page_widgets.cpp
    ${REPO_ROOT}/src/app/page_animation.cpp
    ${REPO_ROOT}/src/app/page_math.cpp
    ${REPO_ROOT}/src/app/page_hdr.cc
    ${REPO_ROOT}/src/app/page_image.cc
    ${REPO_ROOT}/src/app/page_gestures.cc
    ${REPO_ROOT}/src/app/page_textedit.cc
    ${REPO_ROOT}/src/app/page_plot.cc
    ${REPO_ROOT}/src/math/msdf_font_metrics_adapter.cpp
    ${VK_CANVAS}/platform/android/android_platform.cc
    ${APP_SHELL_NATIVE_APP_GLUE_DIR}/android_native_app_glue.c)

add_dependencies(ui_demo compile_shared_shaders compile_canvas_shaders)

target_include_directories(ui_demo PRIVATE
    ${REPO_ROOT}/src
    ${REPO_ROOT}/src/app
    ${APP_SHELL_DIR}
    ${APP_SHELL_DIR}/os
    ${VK_CANVAS}/core
    ${VK_CANVAS}/first_party/vulkan_font_engine/core
    ${VK_CANVAS}/platform/android
    ${APP_SHELL_NATIVE_APP_GLUE_DIR}
    ${_gen_dir})

target_compile_options(ui_demo PRIVATE $<$<COMPILE_LANGUAGE:CXX>:-O2>)

target_link_libraries(ui_demo PRIVATE
    app_shell_android
    vk_canvas_core
    mathcore
    android
    log
    vulkan)

# NativeActivity's entry point is found by name at dlopen time; without this the
# linker drops it as unreferenced and the activity starts into nothing.
target_link_options(ui_demo PRIVATE -u ANativeActivity_onCreate)
